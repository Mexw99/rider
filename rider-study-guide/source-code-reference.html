4. Road Matrix Service: Validate, OSRM และ Cache

ไฟล์จริง: backend/services/road-matrix.ts

const cache = new Map<string, {
  expires: number;
  matrix: RoadDistanceMatrix
}>();

const CACHE_TTL_MS = 60 * 60 * 1000;

export function parseRoadPoints(value: unknown): RoadCoordinate[] {
  if (!Array.isArray(value) || value.length < 2 || value.length > 31) {
    throw new RoadMatrixError(400,
      "ระยะทางตามถนนรองรับร้านและจุดส่ง 1–30 จุด");
  }

  return value.map((point: unknown) => {
    const { latitude, longitude } =
      point as Record<string, unknown>;

    if (typeof latitude !== "number" ||
        typeof longitude !== "number" ||
        !Number.isFinite(latitude) ||
        !Number.isFinite(longitude) ||
        Math.abs(latitude) > 90 ||
        Math.abs(longitude) > 180) {
      throw new RoadMatrixError(400,
        "พิกัดสำหรับคำนวณถนนไม่ถูกต้อง");
    }

    return { latitude, longitude };
  });
}
export async function getRoadDistanceMatrix(
  input: readonly RoadCoordinate[]
): Promise<RoadDistanceMatrix> {
  const points = parseRoadPoints(input);
  const key = JSON.stringify(points);
  const cached = cache.get(key);

  if (cached && cached.expires > Date.now()) {
    return structuredClone(cached.matrix);
  }

  const coordinates = points
    .map(point => \`\${point.longitude},\${point.latitude}\`)
    .join(";");

  const response = await fetch(
    \`https://router.project-osrm.org/table/v1/driving/\${coordinates}?annotations=distance\`,
    { signal: AbortSignal.timeout(8000) }
  );

  const data = await response.json();
  const distancesKm = data.distances.map(
    (row: number[]) => row.map(distance => distance / 1000)
  );

  const matrix = {
    points,
    distancesKm,
    distanceBasis: "road" as const
  };

  cache.set(key, {
    expires: Date.now() + CACHE_TTL_MS,
    matrix
  });

  return matrix;
}

จุดสำคัญ:

  • OSRM ใช้พิกัดรูปแบบ longitude,latitude ใน URL
  • OSRM คืนระยะทางเป็นเมตร จึงหาร 1000 เป็นกิโลเมตร
  • Matrix เป็นตารางระยะทางทุกคู่ของจุด
  • Cache ลดการเรียก OSRM ซ้ำเมื่อพิกัดชุดเดิม
  • Cache เป็นหน่วยความจำของ Process ไม่ใช่ข้อมูลถาวรใน PostgreSQL

7. Route Distance: สูตร Haversine ในโค้ดจริง

ไฟล์จริง: frontend/src/app/pages/delivery-routes/planning/route-distance.ts

const EARTH_RADIUS_KM = 6371;

function toRadians(degrees: number): number {
  return degrees * Math.PI / 180;
}

export function calculateDistanceKm(
  start: Coordinate,
  end: Coordinate
): number {
  const latitude1 = toRadians(start.latitude);
  const latitude2 = toRadians(end.latitude);
  const deltaLatitude =
    toRadians(end.latitude - start.latitude);
  const deltaLongitude =
    toRadians(end.longitude - start.longitude);

  const a =
    Math.sin(deltaLatitude / 2) ** 2 +
    Math.cos(latitude1) *
    Math.cos(latitude2) *
    Math.sin(deltaLongitude / 2) ** 2;

  const limitedA = Math.min(1, Math.max(0, a));
  const centralAngle =
    2 * Math.atan2(
      Math.sqrt(limitedA),
      Math.sqrt(1 - limitedA)
    );

  return EARTH_RADIUS_KM * centralAngle;
}

สูตรนี้ใช้ดูระยะห่างเชิงภูมิศาสตร์ ไม่ใช่เส้นทางรถจริง ในระบบจึงใช้เป็นพื้นฐาน/ทางสำรอง ส่วนแผนถนนจริงใช้ OSRM

8. Route Distance: รวมร้าน → ลูกค้า → กลับร้าน

export function calculateRouteDistance(
  store: Coordinate,
  orderedOrders: readonly RoutingOrder[]
): RouteDistanceResult {
  const stops = [];
  let previous = store;
  let cumulativeDistanceKm = 0;

  for (let index = 0;
       index < orderedOrders.length;
       index += 1) {
    const order = orderedOrders[index];
    const distanceFromPreviousKm =
      calculateDistanceKm(previous, order);

    cumulativeDistanceKm += distanceFromPreviousKm;

    stops.push({
      ...order,
      sequence: index + 1,
      distanceFromPreviousKm,
      cumulativeDistanceKm,
    });

    previous = order;
  }

  const returnDistanceKm =
    orderedOrders.length === 0
      ? 0
      : calculateDistanceKm(previous, store);

  return {
    stops,
    returnDistanceKm,
    totalDistanceKm:
      cumulativeDistanceKm + returnDistanceKm,
  };
}

ตัวแปร previous ทำให้ระบบคำนวณจากจุดก่อนหน้าไปจุดถัดไป ไม่ได้วัดทุกจุดกลับร้านเสมอ ส่วน returnDistanceKm คำนวณเฉพาะช่วงท้ายกลับร้าน

9. Route Timing: เวลาออก 11:30 และ Deadline 12:30

ไฟล์จริง: frontend/src/app/pages/delivery-routes/planning/route-timing.ts

export function calculateRouteTiming(
  route: RouteDistanceResult,
  departureMinute = 11 * 60 + 30,
  deadlineMinute = 12 * 60 + 30,
  speedKmh = 30,
): RouteTimingResult {
  const stops = route.stops.map(stop => {
    const elapsedMinutes =
      stop.cumulativeDistanceKm / speedKmh * 60;

    const arrivalMinute =
      departureMinute + elapsedMinutes;

    const lateMinutes =
      Math.max(0, arrivalMinute - deadlineMinute);

    return {
      ...stop,
      arrivalMinute,
      lateMinutes,
      isLate: lateMinutes > 0,
    };
  });

  return {
    stops,
    totalTravelMinutes:
      route.totalDistanceKm / speedKmh * 60,
    lastDeliveryMinute:
      stops.at(-1)?.arrivalMinute ?? null,
    returnMinute:
      departureMinute + route.totalDistanceKm / speedKmh * 60,
    lateOrderCount:
      stops.filter(stop => stop.isLate).length,
  };
}

11:30 ถูกแทนด้วย 690 นาทีจากเที่ยงคืน และ 12:30 คือ 750 นาที การถึงตรง 750 ยังไม่ถือว่าสาย เพราะตรวจ lateMinutes > 0

10. Route Cost: สูตรต้นทุนในโค้ดจริง

ไฟล์จริง: frontend/src/app/pages/delivery-routes/planning/route-cost.ts

export const COST_RULES = {
  pricePerBox: 65,
  foodCostPerBox: 40,
  dispatchCostPerTrip: 15,
  transportCostPerKmPerBox: 2,
  latePenaltyPerOrder: 20,
} as const;

const revenue =
  trip.totalBoxes * COST_RULES.pricePerBox;

const foodCost =
  trip.totalBoxes * COST_RULES.foodCostPerBox;

const dispatchCost =
  COST_RULES.dispatchCostPerTrip;

const transportCost =
  distanceKm *
  trip.totalBoxes *
  COST_RULES.transportCostPerKmPerBox;

const estimatedPenalty =
  trip.timing.lateOrderCount *
  COST_RULES.latePenaltyPerOrder;

const totalCost =
  foodCost + dispatchCost +
  transportCost + estimatedPenalty;

return {
  revenue,
  totalCost,
  profit: revenue - totalCost
};

    route_stop.rider_id,
    route_plan.worksheet_code,

    order_row.box_quantity
  FROM route_stops AS route_stop

       route_stop.route_plan_id
  INNER JOIN orders AS order_row
    ON order_row.order_id = route_stop.order_id
  WHERE route_plan.route_date = BUSINESS_DATE
    AND route_plan.status = "SELECTED"
)
SELECT
  rider.rider_id,
  rider.name,
  MAX(today_stops.worksheet_code),
  COUNT(today_stops.rider_id) AS order_count,
  SUM(today_stops.box_quantity) AS total_box_count
FROM riders AS rider
LEFT JOIN today_stops
  ON today_stops.rider_id = rider.rider_id
GROUP BY rider.rider_id, rider.name;

26. เปลี่ยนผลการส่ง

protected async updateStopStatus(
  stop: RiderWorkStop,
  status: RiderWorkStop["status"]
): Promise<void> {
  if (this.statusUpdating() !== null ||
      stop.status === status) return;

  this.statusUpdating.set(stop.routeStopId);

  try {
    await this.workApi.updateStopStatus(
      this.auth.getWorksheetCode(),
      stop.routeStopId,
      status
    );

    this.work.update(current => ({
      ...current,
      stops: current.stops.map(item =>
        item.routeStopId === stop.routeStopId
          ? { ...item, status }
          : item
      )
    }));
  } finally {
    this.statusUpdating.set(null);
  }
}

Frontend ส่ง routeStopId และสถานะ แต่ Backend ต้องตรวจซ้ำว่า Stop นั้นอยู่ใน Worksheet ของ Rider ก่อน Update
